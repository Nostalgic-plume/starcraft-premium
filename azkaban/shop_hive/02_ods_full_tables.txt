-- ============================================================
-- ODS 层 — 全量表 DDL（10 张）
-- 分区类型：PARTITIONED BY (dt STRING) — 每天全量覆盖
-- 数据格式：TEXTFILE，字段分隔符 '\001'（与 Sqoop 导入一致）
-- 数据来源：MySQL shop 数据库，通过 Sqoop 全量导入
-- ============================================================

-- 使用 MR 引擎建表（兼容性最佳）
SET hive.execution.engine=mr;

-- ============================================================
-- 1. ods_address_full：地址表
--    对应 MySQL：address
--    全量覆盖：每次导入完整数据至当天分区
-- ============================================================
DROP TABLE IF EXISTS ods.ods_address_full;
CREATE EXTERNAL TABLE ods.ods_address_full (
    id             BIGINT       COMMENT '地址ID',
    user_id        BIGINT       COMMENT '用户ID',
    receiver_name  STRING       COMMENT '收货人姓名',
    receiver_phone STRING       COMMENT '收货人电话',
    province       STRING       COMMENT '省份',
    city           STRING       COMMENT '城市',
    district       STRING       COMMENT '区县',
    detail         STRING       COMMENT '详细地址',
    is_default     TINYINT      COMMENT '是否默认地址：0否 1是',
    create_time    STRING       COMMENT '创建时间',
    update_time    STRING       COMMENT '更新时间',
    deleted        TINYINT      COMMENT '逻辑删除：0未删 1已删'
) PARTITIONED BY (dt STRING COMMENT '分区：数据日期')
ROW FORMAT DELIMITED FIELDS TERMINATED BY '\001'
STORED AS TEXTFILE
LOCATION '/warehouse/ods/ods_address_full';
